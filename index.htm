    <script>

        let gameState = {

            team1: "Judá",

            team2: "Israel",

            score1: 0,

            score2: 0,

            target: 3,

            attacker: 1,

            ballX: 13,

            ballY: 8,

            timer: 20,

            interval: null,

            currentCard: null

        };

 

        const sampleCards = [

            { ref: "A1", q: "Palavra chave no livro de Gênesis.", correct: "No princípio", options: ["No princípio", "Criação divina", "No princípio crió Deus", "A terra era sem forma"], refBib: "Gn 1:1", dir: "A", mov: 3 },

            { ref: "A1", q: "Bisavó de Noé. Não conheceu a morte.", correct: "Enoque", options: ["Matusalém", "Enoque", "Jared", "Lameque"], refBib: "Gn 5:21-29", dir: "ADD", mov: 4 },

            { ref: "A1", q: "Primeira promessa messiânica.", correct: "Gênesis 3:15", options: ["Gênesis 3:15", "Isaías 7:14", "Gênesis 12:3", "Salmo 22:1"], refBib: "Gn 3:15", dir: "A", mov: 5 },

            { ref: "A2", q: "Lugar onde Samuel orou por Israel e o julgou.", correct: "Mizpá", options: ["Betel", "Mizpá", "Ramá", "Silo"], refBib: "1 Sm 7:5,6", dir: "ADE", mov: 3 },

            { ref: "A2", q: "Começou a reinar em Jerusalém desde os 7 anos.", correct: "Joás", options: ["Josias", "Joás", "Uzias", "Salomão"], refBib: "2 Rs 11:21", dir: "A", mov: 4 },

            { ref: "A3", q: "Quantos anos viveu Adão?", correct: "Viveu 930 anos", options: ["Viveu 930 anos", "Viveu 969 anos", "Viveu 777 anos", "Viveu 500 anos"], refBib: "Gn 5:5", dir: "ADD", mov: 4 },

            { ref: "B1", q: "É conhecido como o Pai da Fé.", correct: "Abraão", options: ["Isaque", "Abraão", "Moisés", "Davi"], refBib: "Rm 4:16", dir: "A", mov: 5 },

            { ref: "B2", q: "Qual o Salmo mais curto?", correct: "Salmo 117", options: ["Salmo 117", "Salmo 133", "Salmo 23", "Salmo 150"], refBib: "Sl 117", dir: "ADE", mov: 3 }

        ];

 

        function startGame() {

            const t1 = document.getElementById('team1-name').value.trim();

            const t2 = document.getElementById('team2-name').value.trim();

            const target = parseInt(document.getElementById('target-goals').value);

           

            if(t1) gameState.team1 = t1;

            if(t2) gameState.team2 = t2;

            if(target) gameState.target = target;

 

            document.getElementById('disp-team1').innerText = gameState.team1;

            document.getElementById('disp-team2').innerText = gameState.team2;

            document.getElementById('score-team1').innerText = gameState.score1;

            document.getElementById('score-team2').innerText = gameState.score2;

 

            document.getElementById('setup-screen').classList.add('hidden');

            document.getElementById('game-screen').classList.remove('hidden');

 

            renderPitch();

            updatePossessionLabel();

        }

 

        function renderPitch() {

            const grid = document.getElementById('pitch-grid');

            grid.innerHTML = '';

            for (let r = 0; r < 17; r++) {

                for (let c = 0; c < 27; c++) {

                    const cell = document.createElement('div');

                    const isDark = (r + c) % 2 === 0;

                    cell.className = `cell ${isDark ? 'cell-dark' : 'cell-light'}`;

                   

                    if (c === gameState.ballX && r === gameState.ballY) {

                        const ball = document.createElement('div');

                        ball.className = 'w-4 h-4 bg-white rounded-full shadow-md border-2 border-slate-900 z-20 flex items-center justify-center text-[8px] font-black text-slate-900';

                        ball.innerText = '⚽';

                        cell.appendChild(ball);

                    }

                    grid.appendChild(cell);

                }

            }

        }

 

        function updatePossessionLabel() {

            const lbl = document.getElementById('disp-possession');

            if (gameState.attacker === 1) {

                lbl.innerText = `${gameState.team1} Atacando ➔`;

                lbl.className = "font-bold text-emerald-400 text-sm bg-slate-900 px-4 py-1 rounded-full border border-emerald-500/30";

            } else {

                lbl.innerText = `⬅️ ${gameState.team2} Atacando`;

                lbl.className = "font-bold text-blue-400 text-sm bg-slate-900 px-4 py-1 rounded-full border border-blue-500/30";

            }

        }

 

        function drawCard() {

            document.getElementById('draw-card-btn').disabled = true;

            const cardPanel = document.getElementById('card-panel');

            cardPanel.classList.remove('hidden');

            document.getElementById('card-result').classList.add('hidden');

 

            const card = sampleCards[Math.floor(Math.random() * sampleCards.length)];

            gameState.currentCard = card;

 

            document.getElementById('card-ref').innerText = `REF: ${card.ref} | ${card.refBib}`;

            document.getElementById('card-question').innerText = card.q;

 

            const optContainer = document.getElementById('options-container');

            optContainer.innerHTML = '';

           

            const shuffledOptions = [...card.options].sort(() => Math.random() - 0.5);

           

            shuffledOptions.forEach(opt => {

                const btn = document.createElement('button');

                btn.className = "w-full text-left bg-slate-900 hover:bg-slate-700 text-slate-200 p-2.5 rounded-lg border border-slate-700 transition text-sm font-medium";

                btn.innerText = opt;

                btn.onclick = () => evaluateAnswer(opt === card.correct);

                optContainer.appendChild(btn);

            });

 

            gameState.timer = 20;

            document.getElementById('timer-display').innerText = `${gameState.timer}s`;

           

            if (gameState.interval) clearInterval(gameState.interval);

            gameState.interval = setInterval(() => {

                gameState.timer--;

                document.getElementById('timer-display').innerText = `${gameState.timer}s`;

                if (gameState.timer <= 0) {

                    clearInterval(gameState.interval);

                    evaluateAnswer(false);

                }

            }, 1000);

        }

 

        function evaluateAnswer(isCorrect) {

            clearInterval(gameState.interval);

            const optContainer = document.getElementById('options-container');

            optContainer.innerHTML = '';

 

            const resultDiv = document.getElementById('card-result');

            resultDiv.classList.remove('hidden');

 

            const defenderName = gameState.attacker === 1 ? gameState.team2 : gameState.team1;

            const attackerName = gameState.attacker === 1 ? gameState.team1 : gameState.team2;

 

            if (isCorrect) {

                resultDiv.className = "text-sm font-bold text-center py-2 rounded-lg bg-emerald-500/20 text-emerald-300 border border-emerald-500/40";

                resultDiv.innerText = `¡Correcto! ${defenderName} defendió e interceptó el balón. ¡Robo de posesión!`;

                gameState.attacker = gameState.attacker === 1 ? 2 : 1;

            } else {

                resultDiv.className = "text-sm font-bold text-center py-2 rounded-lg bg-rose-500/20 text-rose-300 border border-rose-500/40";

                resultDiv.innerText = `Incorrecto / Tiempo agotado. ${attackerName} avanza con el balón (${gameState.currentCard.dir}, ${gameState.currentCard.mov} casillas).`;

               

                if (gameState.attacker === 1) {

                    gameState.ballX = Math.min(26, gameState.ballX + gameState.currentCard.mov);

                    if (gameState.ballX >= 26) {

                        gameState.score1++;

                        document.getElementById('score-team1').innerText = gameState.score1;

                        alert(`¡GOOOOL de ${gameState.team1}! ⚽🎉`);

                        gameState.ballX = 13;

                        gameState.ballY = 8;

                    }

                } else {

                    gameState.ballX = Math.max(0, gameState.ballX - gameState.currentCard.mov);

                    if (gameState.ballX <= 0) {

                        gameState.score2++;

                        document.getElementById('score-team2').innerText = gameState.score2;

                        alert(`¡GOOOOL de ${gameState.team2}! ⚽🎉`);

                        gameState.ballX = 13;

                        gameState.ballY = 8;

                    }

                }

                renderPitch();

            }

 

            updatePossessionLabel();

 

            setTimeout(() => {

                document.getElementById('card-panel').classList.add('hidden');

                document.getElementById('draw-card-btn').disabled = false;

            }, 3500);

        }

    </script>
